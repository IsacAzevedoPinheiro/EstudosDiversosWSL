SELECT r.RouteID, r.Origin, r.Destination
FROM route as r
JOIN airport as a ON r.Origin = a.AirportID
WHERE a.AirportCode = 'LHR'
OR a.AirportCode = 'AMS';

SELECT r.RouteID, r.Origin, r.Destination
FROM route as r
JOIN airport as a ON r.Origin = a.AirportID
WHERE a.AirportCode = 'LHR'
UNION
SELECT r.RouteID, r.Origin, r.Destination
FROM route as r
JOIN airport as a ON r.Origin = a.AirportID
WHERE a.AirportCode = 'AMS';

SELECT r.RouteID, a.AirportCode as Origem, ai.AirportCode as Destino
FROM route as r
JOIN airport as a ON r.Origin = a.AirportID
JOIN airport as ai ON r.Destination = ai.AirportID
ORDER BY Origem, Destino; --Cuidado com a ordem dos apelidos
